% === Custom commands "modern pitch deck" ===============================

% Hero stat — gigante number on slide
\newcommand{\herostat}[2]{%
  \vspace*{\fill}
  \begin{center}
  {\fontsize{96pt}{108pt}\selectfont\bfseries\color{accent}#1}\\[0.5em]
  {\Large\color{textmuted}#2}
  \end{center}
  \vspace*{\fill}
}

% Big idea — large centered text
\newcommand{\bigidea}[1]{%
  \vspace*{0.3em}
  \begin{center}
  {\fontsize{32pt}{40pt}\selectfont\bfseries\color{textmain}#1}
  \end{center}
  \vspace*{0.3em}
}

% Sub-hero — secondary big text
\newcommand{\subhero}[1]{%
  \begin{center}
  {\LARGE\bfseries\color{accent}#1}
  \end{center}
}

% Stat card — small card with big number + label
\newcommand{\statcard}[2]{%
  \begin{tcolorbox}[enhanced, colback=bgcard, colframe=bgcard, arc=5pt, boxrule=0pt,
    width=3.4cm, halign=center, valign=center,
    height=2.2cm, fontupper=\color{textmain}]
    {\fontsize{32pt}{38pt}\selectfont\bfseries\color{accent}#1}\\[0.2em]
    {\footnotesize\color{textmuted}#2}
  \end{tcolorbox}
}

% Pill / chip badge
\newcommand{\pill}[2]{%
  \tikz[baseline=(b.base)]\node[fill=#1, text=textmain, rounded corners=8pt,
    inner sep=4pt, font=\footnotesize\bfseries] (b) {#2};%
}
